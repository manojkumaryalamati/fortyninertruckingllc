rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
    }

    function validDriverApp() {
      return request.resource.data.keys().hasAll([
          "firstName","lastName","email","phone","yearsCommercialDriving","cdlNumber"
        ])
        && request.resource.data.keys().hasOnly([
          "firstName","lastName","email","phone","yearsCommercialDriving","cdlNumber",
          "resumeUrl","resumePath","createdAt","status"
        ])
        && request.resource.data.firstName is string
        && request.resource.data.lastName is string
        && request.resource.data.email is string
        && (request.resource.data.phone is string || request.resource.data.phone is int)
        && request.resource.data.yearsCommercialDriving is string
        && request.resource.data.cdlNumber is string
        && (!("resumeUrl" in request.resource.data) || request.resource.data.resumeUrl is string)
        && (!("resumePath" in request.resource.data) || request.resource.data.resumePath is string)
        // status optional; if provided must be "new"
        && (!("status" in request.resource.data) || request.resource.data.status == "new")
        // createdAt optional; if provided must be timestamp
        && (!("createdAt" in request.resource.data) || request.resource.data.createdAt is timestamp);
    }

    // ✅ Driver Applications (public create, admin manages)
    match /driver_applications/{appId} {
      allow create: if validDriverApp();
      allow read, update, delete: if isAdmin();
    }

    // ✅ If your code uses camelCase collection name, allow that too
    match /driverApplications/{appId} {
      allow create: if validDriverApp();
      allow read, update, delete: if isAdmin();
    }

    // Contact submissions: public create only, admin manages
    match /contact_submissions/{submissionId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    // Admin-only collections
    match /trucks/{truckId} { allow read, write: if isAdmin(); }
    match /drivers/{driverId} { allow read, write: if isAdmin(); }
    match /trips/{tripId} { allow read, write: if isAdmin(); }
    match /documents/{docId} { allow read, write: if isAdmin(); }

    match /users/{userId} {
      allow read: if (isSignedIn() && request.auth.uid == userId) || isAdmin();
      allow write: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
